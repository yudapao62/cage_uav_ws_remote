cmake_minimum_required(VERSION 3.0.2)
project(cage_web_bridge)
find_package(catkin REQUIRED COMPONENTS cage_msgs rospy std_msgs)
catkin_package(CATKIN_DEPENDS cage_msgs rospy std_msgs)
catkin_install_python(PROGRAMS scripts/web_bridge_node.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY www config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test)
endif()
