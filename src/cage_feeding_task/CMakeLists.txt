cmake_minimum_required(VERSION 3.0.2)
project(cage_feeding_task)
find_package(catkin REQUIRED COMPONENTS cage_grid cage_msgs mavros_msgs rospy sensor_msgs std_msgs)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS cage_grid cage_msgs mavros_msgs rospy sensor_msgs std_msgs)
catkin_install_python(PROGRAMS scripts/feeding_task_manager_node.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test)
endif()
