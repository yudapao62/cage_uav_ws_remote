cmake_minimum_required(VERSION 3.0.2)
project(cage_waypoint_planner)
find_package(catkin REQUIRED COMPONENTS rospy sensor_msgs cage_msgs cage_geo cage_grid)
catkin_python_setup()

catkin_install_python(PROGRAMS scripts/waypoint_planner_node.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test)
endif()
